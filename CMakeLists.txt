cmake_minimum_required(VERSION 3.24)
project(VYNE VERSION 0.3.2 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "VYNE requires Windows x64")
endif()
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(VYNE SHARED src/bootstrap.cpp src/native_mode_probe.cpp src/export_menu.cpp src/export_settings.cpp src/native_output.cpp src/builds.rc)
add_library(vyne_graphics_mode STATIC src/graphics_mode.cpp)
target_compile_features(vyne_graphics_mode PUBLIC cxx_std_17)
target_compile_features(VYNE PRIVATE cxx_std_17)
target_compile_definitions(VYNE PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(VYNE PRIVATE version user32 ole32 shell32 uuid minhook vyne_graphics_mode)
set_target_properties(VYNE PROPERTIES PREFIX "" SUFFIX ".asi")
if(MSVC)
  target_compile_options(VYNE PRIVATE /W4 /WX /EHsc)
  set_property(TARGET VYNE PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET vyne_graphics_mode PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET minhook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(graphics_mode_test tests/graphics_mode_test.cpp)
  target_link_libraries(graphics_mode_test PRIVATE vyne_graphics_mode)
  if(MSVC)
    set_property(TARGET graphics_mode_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME graphics_mode_transaction COMMAND graphics_mode_test)
  add_executable(export_menu_test tests/export_menu_test.cpp src/export_settings.cpp)
  target_compile_features(export_menu_test PRIVATE cxx_std_17)
  target_compile_definitions(export_menu_test PRIVATE UNICODE _UNICODE)
  target_include_directories(export_menu_test PRIVATE vendor/minhook/include)
  target_link_libraries(export_menu_test PRIVATE ole32 shell32 uuid)
  if(MSVC)
    target_compile_options(export_menu_test PRIVATE /utf-8 /W4 /WX /EHsc)
    set_property(TARGET export_menu_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME export_menu_model COMMAND export_menu_test)
  add_executable(native_output_test tests/native_output_test.cpp src/native_output.cpp)
  target_compile_features(native_output_test PRIVATE cxx_std_17)
  target_compile_definitions(native_output_test PRIVATE UNICODE _UNICODE)
  target_link_libraries(native_output_test PRIVATE ole32 shell32 uuid)
  if(MSVC)
    target_compile_options(native_output_test PRIVATE /utf-8 /W4 /WX /EHsc)
    set_property(TARGET native_output_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME native_output_delivery COMMAND native_output_test)
endif()
