cmake_minimum_required(VERSION 3.24)
project(VYNE VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "VYNE requires Windows x64")
endif()
add_library(VYNE SHARED src/bootstrap.cpp)
target_compile_features(VYNE PRIVATE cxx_std_17)
target_compile_definitions(VYNE PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(VYNE PRIVATE version)
set_target_properties(VYNE PROPERTIES PREFIX "" SUFFIX ".asi")
if(MSVC)
  target_compile_options(VYNE PRIVATE /W4 /WX /EHsc)
  set_property(TARGET VYNE PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
